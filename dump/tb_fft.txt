`timescale 1ns / 1ps

module tb_fft_top;

// Parameters & Signals
localparam N = `N;
localparam W = `DATA_WIDTH;
localparam F = `FRACTIONAL_BITS;

localparam CLK_PERIOD = 10;
localparam M = 8;   // length of input stream

// Calculate pipeline latency to ignore the flush garbage
// Latency = Delay Buffers (N-1) + Pipeline Registers (clog2(N))
localparam PIPELINE_LATENCY = (N - 1) + $clog2(N);

// 1.0 in Q-format fixed-point 
localparam signed [W-1:0] ONE_FP = (1 << F); 

reg clk;
reg reset_n;

reg valid_in;
reg signed [W-1:0] in_real, in_imag;

wire valid_out;
wire signed [W-1:0] out_real, out_imag;

// DUT
top_level #(.N(N), .W(W), .F(F)) dut (
    .clk(clk), .reset_n(reset_n), .en(valid_in),
    .in_real(in_real), .in_imag(in_imag),
    .valid_out(valid_out), .out_real(out_real), .out_imag(out_imag)
);

// clock
initial begin
    clk = 0;
    forever #(CLK_PERIOD/2) clk = ~clk;
end

// Arrays for stimulus and golden comparison
reg signed [W-1:0] input_mem [0:M-1];
real golden_real [0:N-1];
real golden_imag [0:N-1];

integer i;
integer fd; // file descriptor

// Helper Function: Bit Reversal for Output Matching
function integer bit_reverse;
    input integer val;
    input integer num_bits;
    integer b, rev;
    begin
        rev = 0;
        for (b = 0; b < num_bits; b = b + 1) begin
            rev = (rev << 1) | ((val >> b) & 1'b1);
        end
        bit_reverse = rev;
    end
endfunction

// Main Stimulus
initial begin
    // Read input fixed-point file
    $readmemh(`INPUT_STREAM_FILE, input_mem);
    
    // Read golden floating-point file (Format: real_value imag_value)
    fd = $fopen("golden_output.txt", "r");
    if (fd == 0) $display("WARNING: Could not open golden_output.txt");
    for (i = 0; i < N; i = i + 1) begin
        $fscanf(fd, "%f %f\n", golden_real[i], golden_imag[i]);
    end
    $fclose(fd);

    // Initialize signals
    reset_n = 0;
    valid_in = 0;
    in_real = 0;
    in_imag = 0;

    // VCD Dump
    $dumpfile("fft_n8.vcd");
    $dumpvars(0, tb_fft_top);

    // Apply Reset
    #20;
    reset_n = 1;
    #10;

    $display("--- Starting N=%0d FFT Injection ---", N);

    // PHASE 1: Inject Data Samples
    for (i = 0; i < M; i = i + 1) begin
        @(posedge clk);
        valid_in = 1;
        in_real = input_mem[i];
        in_imag = 0;
    end

    // PHASE 2: Flush the Pipeline
    for (i = 0; i < M + 2; i = i + 1) begin
        @(posedge clk);
        valid_in = 1;
        in_real = 0;
        in_imag = 0;
    end

    // PHASE 3: Stop Streaming
    @(posedge clk);
    valid_in = 0;
    
    #50;
    $display("--- Simulation Complete ---");
    $finish;
end

// -------------------------------------------------------------------------
// Output Monitor & SQNR Calculator
// -------------------------------------------------------------------------
integer out_count = 1;
integer valid_data_count = 0;
integer bin_idx;

// Power accumulators for SQNR
real signal_power = 0.0;
real noise_power = 0.0;
real expected_real, expected_imag;
real error_real, error_imag;
real sqnr_db;

always @(posedge clk) begin
    if (valid_out) begin
        
        // Only evaluate data AFTER the pipeline flush period
        if (out_count > PIPELINE_LATENCY && valid_data_count < N) begin
            
            // 1. Convert fixed-point hardware output to scaled float
            real real_float, imag_float;
            real_float = $itor(out_real) / (2.0 ** F) * N;
            imag_float = $itor(out_imag) / (2.0 ** F) * N;

            // 2. Map current sequential output to its bit-reversed frequency bin
            bin_idx = bit_reverse(valid_data_count, $clog2(N));

            // 3. Fetch the sequential Golden Reference
            expected_real = golden_real[bin_idx];
            expected_imag = golden_imag[bin_idx];

            // 4. Calculate Quantization Error
            error_real = expected_real - real_float;
            error_imag = expected_imag - imag_float;

            // 5. Accumulate Power
            signal_power = signal_power + (expected_real * expected_real) + (expected_imag * expected_imag);
            noise_power  = noise_power  + (error_real * error_real) + (error_imag * error_imag);

            $display("Bin %0d | HW: (%f, %f) | Target: (%f, %f)", 
                        bin_idx, real_float, imag_float, expected_real, expected_imag);
            
            valid_data_count = valid_data_count + 1;

            // Print Final SQNR when frame is complete
            if (valid_data_count == N) begin
                if (noise_power > 0.0) begin
                    sqnr_db = 10.0 * $log10(signal_power / noise_power);
                    $display("========================================");
                    $display("RTL SQNR Result: %f dB", sqnr_db);
                    $display("========================================");
                end else begin
                    $display("RTL SQNR Result: INFINITY (Perfect Match)");
                end
            end

        end
        out_count <= out_count + 1;
    end
end

endmodule
